USE `superadmin_db`;
SET NAMES utf8mb4;
SET FOREIGN_KEY_CHECKS = 0;

-- #1: points_mall_items 添加 points_required 列
SET @col_exists := (
  SELECT COUNT(*) FROM information_schema.COLUMNS
  WHERE TABLE_SCHEMA = DATABASE() AND TABLE_NAME = 'points_mall_items'
    AND COLUMN_NAME = 'points_required'
);
SET @sql := IF(@col_exists = 0,
  'ALTER TABLE `points_mall_items`
     ADD COLUMN `points_required` INT UNSIGNED NOT NULL DEFAULT 0 COMMENT ''商品原价积分'' AFTER `cost_points`',
  'SELECT "points_mall_items.points_required already exists, skip add"'
);
PREPARE stmt FROM @sql; EXECUTE stmt; DEALLOCATE PREPARE stmt;

-- #2: 将现有商品的 points_required 设置为 cost_points 的值
UPDATE `points_mall_items` SET `points_required` = `cost_points` WHERE `points_required` = 0;

SET FOREIGN_KEY_CHECKS = 1;

-- 验证
SELECT COUNT(*) AS points_required_non_zero FROM `points_mall_items` WHERE `points_required` > 0;

- [x] **Step 1: 添加获取会员折扣率的辅助方法**

在 `PointsMallService` 类中添加：

```typescript
/** 获取会员商城折扣率（默认 1.00 = 无折扣） */
private async getMemberDiscount(userId: number): Promise<number> {
  const member: any = await this.ctx.model.UserMember.findOne({
    where: { userId },
    include: [{ model: this.ctx.model.MemberLevel, as: 'level' }],
  });
  if (!member) return 1.00;
  const benefits = member.level?.benefits;
  if (benefits && typeof benefits === 'object') {
    return parseFloat(benefits.discount ?? 1.00);
  }
  return 1.00;
}

/** 计算实际支付积分（应用会员折扣，向下取整，最低1积分） */
private calcActualPoints(pointsRequired: number, discount: number): number {
  if (discount >= 1.00) return pointsRequired;
  const result = Math.floor(pointsRequired * discount);
  return result > 0 ? result : 1;  // 折扣后至少为 1 积分
}
```

async listItems(filter: { category?: string; userId?: number } = {}) {
  // ... 原有查询逻辑 ...

  // 如果传入 userId，计算会员折扣
  let discount = 1.00;
  if (filter.userId) {
    discount = await this.getMemberDiscount(filter.userId);
  }

  return items.map((item: any) => {
    const itemData = item.toJSON();
    const pointsRequired = itemData.pointsRequired || itemData.costPoints;
    const pointsActual = filter.userId
      ? this.calcActualPoints(pointsRequired, discount)
      : pointsRequired;

    return {
      ...itemData,
      pointsRequired,
      pointsActual,
      costPoints: pointsActual,  // 保持兼容
    };
  });
}
