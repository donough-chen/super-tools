## 一、背景与目标

### 1.1 背景

Spec-A1 / Spec-A2 / Spec-B 已完成后端 RBAC + 审计 + 反馈/统计接口；管理端前端项目（`super-tools-admin`）已由 Spec-B 搭建好脚手架（路由占位 / dva models / AuthWrapper / AuthButton）。

进入 Spec-C 前端业务页落地阶段。Spec-C 的 9 个页面规模超大（30+ task），按"消费方"拆分为：

| 子 spec | 页面 | 后端来源 |
|---|---|---|
| **Spec-C1（本文）** | system 4 页：roles / permissions / audit-logs / permission-test | Spec-A1 + Spec-B |
| Spec-C2 | user / feedback / member 3 页 | Spec-A2 + Spec-B |
| Spec-C3 | stats 4 页 + dashboard 接通 | Spec-A2 |

### 1.2 目标

落地管理端**系统管理 4 页**：

1. **角色管理** `/system/roles` — 列表 / CRUD / 树形赋权
2. **权限管理** `/system/permissions` — 只读浏览（树形 + 详情）
3. **审计日志** `/system/audit-logs` — 列表（7 维筛选）+ 详情（双列 JSON + diff 表）+ CSV 导出
4. **权限测试** `/system/permission-test` — Tab 切换 3 mode（user-overview / user-check / role-check）

### 1.3 不在本 spec 范围（YAGNI）

- ❌ 权限管理页的增删改（Q3 决策）
- ❌ AuditLogs 列表轮询自动刷新（O2）
- ❌ PermissionTest URL state 同步（O3）
- ❌ 通知中心 / 邮件报警
- ❌ 审计字段对比的高亮 diff 视图（B+ 完整版）
- ❌ 角色批量操作（克隆、批量启停）
- ❌ 全局字典抽象（4 个 page 内联 const）
- ❌ 页面级集成测试（仅 service / utils 单测）

### 3.6 关键不变量（8 条）

1. **service 层无业务逻辑**：service 函数只做 URL 拼装 + request 调用 + 类型标注
2. **所有写操作按钮挂 `AuthButton`**：检查对应权限码（如 `role:create`）；无权限时按钮隐藏
3. **super_admin 角色防误编辑**：Roles 页编辑/赋权/删除按钮 disabled + tooltip
4. **diffFields 跳过系统字段**：固定黑名单 `['updatedAt', 'createdAt', 'deletedAt']`
5. **CSV 导出走浏览器原生下载**：`responseType:'blob'` + `URL.createObjectURL` + `<a download>`
6. **赋权抽屉只传叶子 IDs**：保存前过滤掉父节点 ID（type=1 顶级 + type=2 中间菜单）
7. **审计列表/详情双层结构遵循后端**：列表无 beforeData/afterData/requestParams；只有详情才有
8. **3 mode Tab 各自独立 state**：切 Tab 不重置已查询结果（React state，不用 URL state）

### 4.5 utils/diffFields.ts

```typescript
/**
 * 字段级 diff 工具
 *
 * - 浅比较两个对象的所有 key
 * - 返回值不同的字段（含 added / removed / changed）
 * - 嵌套对象/数组用 JSON.stringify 比较（不深入展开）
 * - 自动跳过系统时间字段（updatedAt / createdAt / deletedAt）
 *
 * 用于 AuditLogs 详情页"变化字段"对比表
 */

export interface DiffEntry {
  key: string;
  before: any;
  after: any;
}

const SYSTEM_FIELDS = new Set(['updatedAt', 'createdAt', 'deletedAt']);

export function diffFields(
  before: Record<string, any> | null | undefined,
  after: Record<string, any> | null | undefined,
): DiffEntry[] {
  if (!before && !after) return [];

  if (before && !after) {
    return Object.keys(before)
      .filter((k) => !SYSTEM_FIELDS.has(k))
      .map((k) => ({ key: k, before: before[k], after: undefined }));
  }

  if (!before && after) {
    return Object.keys(after)
      .filter((k) => !SYSTEM_FIELDS.has(k))
      .map((k) => ({ key: k, before: undefined, after: after[k] }));
  }

  const allKeys = new Set([
    ...Object.keys(before!),
    ...Object.keys(after!),
  ]);

  const result: DiffEntry[] = [];
  for (const k of allKeys) {
    if (SYSTEM_FIELDS.has(k)) continue;
    const b = before![k];
    const a = after![k];
    if (!isEqual(b, a)) {
      result.push({ key: k, before: b, after: a });
    }
  }
  return result;
}

function isEqual(a: any, b: any): boolean {
  if (a === b) return true;
  if (a == null || b == null) return a === b;
  if (typeof a !== typeof b) return false;
  if (typeof a === 'object') {
    try {
      return JSON.stringify(a) === JSON.stringify(b);
    } catch {
      return false;
    }
  }
  return false;
}
```

```typescript
import type { DataNode } from 'antd/lib/tree';
import type { PermissionTreeNode } from '@/services/permission';
import { Tag, Space } from 'antd';
import React from 'react';

/** PermissionTreeNode → Antd TreeData（带 code Tag + type 标识） */
export function toAntdTreeData(nodes: PermissionTreeNode[]): DataNode[] {
  return nodes.map((n) => ({
    key: n.id,
    title: (
      <Space>
        <Tag>{n.code}</Tag>
        <span>{n.name}</span>
        {n.type === 3 && <Tag color="purple">按钮</Tag>}
        {n.type === 4 && <Tag color="cyan">{n.method}</Tag>}
      </Space>
    ),
    children: n.children?.length ? toAntdTreeData(n.children) : undefined,
  }));
}

/** 从 checkedKeys 过滤出叶子 ID（保存赋权时用） */
export function filterLeafIds(
  tree: PermissionTreeNode[],
  checked: number[],
): number[] {
  const leafIds = new Set<number>();
  const walk = (nodes: PermissionTreeNode[]) => {
    for (const n of nodes) {
      const isLeaf = !n.children || n.children.length === 0;
      if (isLeaf && checked.includes(n.id)) leafIds.add(n.id);
      if (n.children) walk(n.children);
    }
  };
  walk(tree);
  return Array.from(leafIds);
}

/** 统计已勾选的叶子数（按钮文案显示） */
export function countLeaves(tree: PermissionTreeNode[], checked: number[]): number {
  return filterLeafIds(tree, checked).length;
}
```

```tsx
    Promise.all([
      getPermissionTree(),
      getRole(role.id),               // 含 permissionIds
    ]).then(([treeRes, roleRes]) => {
      setPermTree(treeRes.data);
      setCheckedKeys(roleRes.data.permissionIds || []);
      setExpandedKeys(treeRes.data.map(n => n.id));   // 默认展开顶级
      setLoading(false);
    });
  }

handleSave():
  setSaving(true);
  const leafIds = filterLeafIds(permTree, checkedKeys);
  await assignRolePermissions(role!.id, leafIds);
  message.success('赋权成功');
  onSuccess(); onClose();

<Drawer width={600}>
  <Tree
    checkable showLine
    treeData={toAntdTreeData(permTree)}
    checkedKeys={checkedKeys}
    expandedKeys={expandedKeys}
    onCheck={setCheckedKeys}
    onExpand={setExpandedKeys}
  />
</Drawer>
```

#### 5.3.5 export 完整逻辑

```typescript
const handleExport = async () => {
  try {
    const res = await exportAuditLogs(filters);
    // res.response 是完整 fetch response
    const ct = res.response?.headers?.get('content-type');
    if (ct?.startsWith('application/json')) {
      const errText = await res.data.text();
      const err = JSON.parse(errText);
      message.error(err.message || '导出失败');
      return;
    }
    const blob = new Blob([res.data], { type: 'text/csv;charset=utf-8' });
    const url = URL.createObjectURL(blob);
    const a = document.createElement('a');
    a.href = url;
    a.download = `audit-logs-${Date.now()}.csv`;
    a.click();
    URL.revokeObjectURL(url);
    if (res.response?.headers?.get('x-audit-truncated') === 'true') {
      message.warning('结果超过 10000 行已截断；请缩小时间范围再次导出');
    }
  } catch (e: any) {
    message.error(e.message || '导出失败');
  }
};
```

```typescript
import { diffFields } from '@/utils/diffFields';

describe('diffFields', () => {
  it('正确识别 added / removed / changed + 跳过系统字段 + 嵌套对比', () => {
    const before = { name: 'A', age: 18, createdAt: '2026-01-01' };
    const after  = { name: 'A', age: 20, email: 'a@b.com', updatedAt: '2026-05-11' };
    const result = diffFields(before, after);

    expect(result).toContainEqual({ key: 'age', before: 18, after: 20 });
    expect(result).toContainEqual({ key: 'email', before: undefined, after: 'a@b.com' });
    expect(result.find(d => d.key === 'name')).toBeUndefined();
    expect(result.find(d => d.key === 'createdAt')).toBeUndefined();
    expect(result.find(d => d.key === 'updatedAt')).toBeUndefined();

    // destroy 场景
    const destroyResult = diffFields({ id: 1, name: 'A' }, null);
    expect(destroyResult).toEqual([
      { key: 'id', before: 1, after: undefined },
      { key: 'name', before: 'A', after: undefined },
    ]);

    // 双 null
    expect(diffFields(null, null)).toEqual([]);

    // 嵌套对比
    const r2 = diffFields({ tags: ['a', 'b'] }, { tags: ['a', 'b', 'c'] });
    expect(r2).toEqual([{ key: 'tags', before: ['a','b'], after: ['a','b','c'] }]);
  });
});
```
