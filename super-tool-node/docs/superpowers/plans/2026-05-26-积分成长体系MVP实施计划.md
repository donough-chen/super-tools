# 积分与成长体系 MVP 实施计划

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** 在现有会员体系基础上，构建完整的积分获取-累积-消耗-过期生命周期，包含任务中心（事件驱动）、签到系统（连续签到+断签归零）、积分商城（纯虚拟商品）、积分过期（FIFO 365天）、积分抵扣（等级阶梯上限）五大模块。

**Architecture:** 采用"事件驱动 + 任务模板"架构。业务模块通过 `app.messenger` 发出领域事件，TaskService 监听事件并匹配任务条件，达标后用户手动领取奖励。积分过期通过定时任务每日扫描 FIFO 清零。积分商城为纯虚拟商品，系统自动履约。

**Tech Stack:** Egg.js (TypeScript) + Sequelize ORM + MySQL + Redis + node-schedule

---

## 决策摘要

| 决策项         | 选择                                                              |
| -------------- | ----------------------------------------------------------------- |
| 成长值 vs 积分 | 强解耦（分行为独立配置）                                          |
| 任务模型       | 事件驱动 + 任务模板表                                             |
| 商城商品       | 纯虚拟商品（优惠券/会员天数/工具特权/徽章）                       |
| 过期策略       | 严格 FIFO 365天，不可续期                                         |
| 签到断签       | 断签归零，未来可加补签                                            |
| 抵扣上限       | 按等级阶梯：free 5%, silver 10%, gold 20%, diamond 30%, black 50% |
| 每日上限       | 任务类 50积分/天，邀请类 5笔/天，消费类不限                       |

```sql
-- 每次获得积分写入一条批次记录，记录过期时间和剩余可用额度
-- ------------------------------------------------------------
CREATE TABLE IF NOT EXISTS `points_batches` (
  `id`            BIGINT UNSIGNED   NOT NULL AUTO_INCREMENT,
  `user_id`       BIGINT UNSIGNED   NOT NULL COMMENT '用户ID',
  `source`        VARCHAR(50)       NOT NULL COMMENT '来源事件码',
  `original`      INT UNSIGNED      NOT NULL COMMENT '原始获得积分',
  `remaining`     INT UNSIGNED      NOT NULL COMMENT '剩余可用积分',
  `expire_at`     DATETIME          NOT NULL COMMENT '过期时间',
  `is_expired`    TINYINT(1)        NOT NULL DEFAULT 0 COMMENT '是否已过期清零',
  `log_id`        BIGINT UNSIGNED   DEFAULT NULL COMMENT '关联 points_logs.id',
  `created_at`    DATETIME          NOT NULL DEFAULT CURRENT_TIMESTAMP,
  PRIMARY KEY (`id`),
  INDEX `idx_user_expire` (`user_id`, `expire_at`),
  INDEX `idx_user_remaining` (`user_id`, `remaining`),
  INDEX `idx_expire_scan` (`is_expired`, `expire_at`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb4 COLLATE=utf8mb4_unicode_ci COMMENT='积分批次表（FIFO过期）';


-- ------------------------------------------------------------

-- ------------------------------------------------------------

  `description`     VARCHAR(500)      DEFAULT NULL COMMENT '任务描述',
  `category`        ENUM('newbie','daily','weekly','monthly','achievement','activity') NOT NULL COMMENT '任务分类',
  `trigger_event`   VARCHAR(50)       NOT NULL COMMENT '监听的领域事件code',
  `condition`       JSON              DEFAULT NULL COMMENT '条件过滤（JSON）',
  `progress_target` INT UNSIGNED      NOT NULL DEFAULT 1 COMMENT '进度目标值',

  `reset_cycle`     ENUM('once','daily','weekly','monthly','yearly') NOT NULL DEFAULT 'once' COMMENT '重置周期',
  `valid_from`      DATETIME          DEFAULT NULL COMMENT '生效开始时间',
  `valid_to`        DATETIME          DEFAULT NULL COMMENT '生效结束时间',
  `required_level`  VARCHAR(30)       DEFAULT NULL COMMENT '等级门槛',
  `daily_cap_group` VARCHAR(30)       DEFAULT NULL COMMENT '日上限分组',
  `sort`            INT               NOT NULL DEFAULT 0 COMMENT '排序',
  `status`          TINYINT UNSIGNED  NOT NULL DEFAULT 1 COMMENT '状态:0禁用,1启用',
  `created_at`      DATETIME          NOT NULL DEFAULT CURRENT_TIMESTAMP,
  `updated_at`      DATETIME          NOT NULL DEFAULT CURRENT_TIMESTAMP ON UPDATE CURRENT_TIMESTAMP,
  PRIMARY KEY (`id`),
  UNIQUE KEY `uk_code` (`code`),
  INDEX `idx_trigger_event` (`trigger_event`),
  INDEX `idx_category` (`category`),
  INDEX `idx_status` (`status`)

-- 3. 用户任务进度表
-- ------------------------------------------------------------

  `task_code`     VARCHAR(50)       NOT NULL COMMENT '任务编码',
  `cycle_key`     VARCHAR(20)       NOT NULL COMMENT '周期键（once/2026-05-26/2026-W21/2026-05）',
  `progress`      INT UNSIGNED      NOT NULL DEFAULT 0 COMMENT '当前进度',
  `status`        ENUM('pending','completed','claimed','expired') NOT NULL DEFAULT 'pending' COMMENT '状态',
  `completed_at`  DATETIME          DEFAULT NULL COMMENT '完成时间',
  `claimed_at`    DATETIME          DEFAULT NULL COMMENT '领取时间',
  `created_at`    DATETIME          NOT NULL DEFAULT CURRENT_TIMESTAMP,
  `updated_at`    DATETIME          NOT NULL DEFAULT CURRENT_TIMESTAMP ON UPDATE CURRENT_TIMESTAMP,
  PRIMARY KEY (`id`),
  UNIQUE KEY `uk_user_task_cycle` (`user_id`, `task_code`, `cycle_key`),
  INDEX `idx_user_status` (`user_id`, `status`),
  INDEX `idx_task_code` (`task_code`)
```

```typescript
// app/service/event.ts
import BaseService from './base';
import { EventEmitter } from 'events';

/**
 * 领域事件载荷接口
 */
export interface DomainEvent {
	/** 事件码（对应 tasks.trigger_event） */
	eventCode: string;
	/** 触发用户ID */
	userId: number;
	/** 事件数据（用于条件匹配） */
	data?: Record<string, any>;
	/** 事件发生时间 */
	timestamp?: Date;
}

/**
 * 领域事件总线服务
 *
 * 职责：
 * 1. 接收业务模块发出的领域事件
 * 2. 分发给 TaskService 进行任务进度匹配
 * 3. 异步处理，不阻塞主业务流程
 */
export default class EventService extends BaseService {
	private static emitter: EventEmitter = new EventEmitter();

	/**
	 * 发出领域事件
	 * 业务模块调用此方法发出事件，TaskService 会异步处理
	 */
	async emit(event: DomainEvent): Promise<void> {
		const { eventCode, userId, data } = event;
		this.ctx.logger.info(
			`[EventService] emit: ${eventCode}, userId=${userId}, data=${JSON.stringify(data || {})}`,
		);

		// 异步分发给任务服务，不阻塞调用方
		setImmediate(async () => {
			try {
				await this.ctx.service.task.dispatch({
					eventCode,
					userId,
					data: data || {},
					timestamp: event.timestamp || new Date(),
				});
			} catch (err: any) {
				this.ctx.logger.error(
					`[EventService] dispatch error: ${err.message}`,
					err,
				);
			}
		});
	}

	/**
	 * 批量发出事件（用于一个操作触发多个事件的场景）
	 */
	async emitBatch(events: DomainEvent[]): Promise<void> {
		for (const event of events) {
			await this.emit(event);
		}
	}
}
```

```typescript
// app/service/sign.ts
import BaseService from './base';

interface SignResult {
	signDate: string;
	streak: number;
	pointsEarned: number;
	growthEarned: number;
	currentPoints: number;
	currentGrowth: number;
	isLevelUp: boolean;
	streakMilestone?: { days: number; bonusPoints: number; bonusGrowth: number };
}

/**
 * 签到服务
 *
 * 规则：
 * - 每日签到获得基础积分（按等级权益 daily_sign_points）
 * - 签到不给成长值（方案A'矩阵）
 * - 断签归零（连续天数重置为0）
 * - 连续签到里程碑：7天(+20积分)、30天(+100积分,+20成长值)、365天(+500积分,+100成长值)
 */
export default class SignService extends BaseService {
	/**
	 * 执行签到
	 */
	async dailySign(userId: number): Promise<SignResult> {
		const today = new Date().toISOString().slice(0, 10);

		// 1. 防重复签到
		await this.checkAlreadySigned(userId, today);

		// 2. 计算连续签到天数
		const streak = await this.calculateStreak(userId, today);

		// 3. 获取签到奖励积分（按等级权益）
		const member = await this.ctx.model.UserMember.findOne({
			where: { userId },
			include: [{ model: this.ctx.model.MemberLevel, as: 'level' }],
		});
		if (!member) this.ctx.throw(404, '会员记录不存在');
		const memberData = (member as any).toJSON();
		const benefits = memberData.level?.benefits || {};
		const basePoints = benefits.daily_sign_points || 1;
		const baseGrowth = 0; // 方案A'：每日签到不给成长值

		// 4. 检查连续签到里程碑奖励
		const milestone = this.checkStreakMilestone(streak);
		const totalPoints = basePoints + (milestone?.bonusPoints || 0);
		const totalGrowth = baseGrowth + (milestone?.bonusGrowth || 0);

		// 5. 写入签到记录 + 更新会员表 + 发放积分（事务）
		const result = await this.ctx.model.transaction(async (t: any) => {
			// 写入签到记录
			await this.ctx.model.UserSign.create(
				{
					userId,
					signDate: today,
					streak,
					pointsEarned: totalPoints,
					growthEarned: totalGrowth,
				},
				{ transaction: t },
			);

			// 更新 user_members 签到字段
			await this.ctx.model.UserMember.update(
				{
					signStreak: streak,
					lastSignDate: today,
					totalSignDays: this.ctx.model.literal('total_sign_days + 1'),
				} as any,
				{
					where: { userId },
					transaction: t,
				},
			);

			// 发放积分（复用 MemberService.addPoints 的事务逻辑）
			const addResult = await this.addPointsInTransaction(
				userId,
				totalPoints,
				totalGrowth,
				'daily_sign',
				t,
				`每日签到(连续${streak}天)`,
			);

			return addResult;
		});

		// 6. 标记 Redis 防重复
		try {
			const secondsRemaining = Math.max(
				1,
				Math.floor(
					(new Date(today + 'T23:59:59').getTime() - Date.now()) / 1000,
				),
			);
			await this.app.redis.setex(
				`sign:${userId}:${today}`,
				secondsRemaining,
				'1',
			);
		} catch {
			/* ignore */
		}

		// 7. 发出签到事件（触发任务中心）
		await this.ctx.service.event.emit({
			eventCode: 'daily_sign',
			userId,
			data: { streak, signDate: today },
		});

		// 8. 如果达到里程碑，额外发出 sign_streak 事件
		if (milestone) {
			await this.ctx.service.event.emit({
				eventCode: 'sign_streak',
				userId,
				data: { streak },
			});
		}

		return {
			signDate: today,
			streak,
			pointsEarned: totalPoints,
			growthEarned: totalGrowth,
			currentPoints: result.currentPoints,
			currentGrowth: result.currentGrowth,
			isLevelUp: result.isLevelUp,
			streakMilestone: milestone || undefined,
		};
	}

	/**
	 * 获取签到状态（当月日历）
	 */
	async getSignStatus(userId: number) {
		const now = new Date();
		const year = now.getFullYear();
		const month = now.getMonth() + 1;
		const monthStr = `${year}-${String(month).padStart(2, '0')}`;
		const { Op } = require('sequelize');

		// 当月签到记录
		const signs = await this.ctx.model.UserSign.findAll({
			where: {
				userId,
				signDate: {
					[Op.gte]: `${monthStr}-01`,
					[Op.lte]: `${monthStr}-31`,
				},
			},
			order: [['sign_date', 'ASC']],
			raw: true,
		});

		// 当前连续天数
		const member = await this.ctx.model.UserMember.findOne({
			where: { userId },
			attributes: ['signStreak', 'lastSignDate', 'totalSignDays'],
			raw: true,
		});

		const today = now.toISOString().slice(0, 10);
		const isTodaySigned = signs.some((s: any) => s.sign_date === today);

		return {
			today,
			isTodaySigned,
			streak: (member as any)?.signStreak || 0,
			totalSignDays: (member as any)?.totalSignDays || 0,
			monthSigns: signs.map((s: any) => ({
				date: s.sign_date,
				streak: s.streak,
				pointsEarned: s.points_earned,
			})),
			nextMilestone: this.getNextMilestone((member as any)?.signStreak || 0),
		};
	}

	// ==================== 私有方法 ====================

	private async checkAlreadySigned(
		userId: number,
		today: string,
	): Promise<void> {
		// 先查 Redis
		try {
			const signed = await this.app.redis.get(`sign:${userId}:${today}`);
			if (signed) this.ctx.throw(400, '今日已签到');
		} catch (err: any) {
			if (err.status === 400) throw err;
		}

		// Redis 不可用时查数据库
		const existing = await this.ctx.model.UserSign.findOne({
			where: { userId, signDate: today },
		});
		if (existing) this.ctx.throw(400, '今日已签到');
	}

	private async calculateStreak(
		userId: number,
		today: string,
	): Promise<number> {
		const member = await this.ctx.model.UserMember.findOne({
			where: { userId },
			attributes: ['signStreak', 'lastSignDate'],
			raw: true,
		});

		const lastSignDate = (member as any)?.lastSignDate;
		const currentStreak = (member as any)?.signStreak || 0;

		if (!lastSignDate) return 1; // 首次签到

		// 计算昨天的日期
		const yesterday = new Date(today);
		yesterday.setDate(yesterday.getDate() - 1);
		const yesterdayStr = yesterday.toISOString().slice(0, 10);

		// 如果昨天签过，连续+1；否则断签归零，从1开始
		if (lastSignDate === yesterdayStr) {
			return currentStreak + 1;
		}

		// 断签归零
		return 1;
	}

	private checkStreakMilestone(
		streak: number,
	): { days: number; bonusPoints: number; bonusGrowth: number } | null {
		const milestones = [
			{ days: 365, bonusPoints: 500, bonusGrowth: 100 },
			{ days: 30, bonusPoints: 100, bonusGrowth: 20 },
			{ days: 7, bonusPoints: 20, bonusGrowth: 0 },
		];

		for (const m of milestones) {
			if (streak === m.days) return m;
		}
		return null;
	}

	private getNextMilestone(
		currentStreak: number,
	): { days: number; remaining: number } | null {
		const milestones = [7, 30, 365];
		for (const days of milestones) {
			if (currentStreak < days) {
				return { days, remaining: days - currentStreak };
			}
		}
		return null;
	}

	/**
	 * 事务内增加积分（复用 member service 的核心逻辑）
	 */
	private async addPointsInTransaction(
		userId: number,
		points: number,
		growthDelta: number,
		source: string,
		transaction: any,
		remark: string,
	) {
		const member = await this.ctx.model.UserMember.findOne({
			where: { userId },
			lock: transaction.LOCK.UPDATE,
			transaction,
		});
		if (!member) this.ctx.throw(404, '会员记录不存在');
		const m = member as any;

		const newPoints = m.points + points;
		const newTotalPoints = m.totalPoints + (points > 0 ? points : 0);
		const newGrowthValue = m.growthValue + (growthDelta > 0 ? growthDelta : 0);

		await m.update(
			{
				points: newPoints,
				totalPoints: newTotalPoints,
				growthValue: newGrowthValue,
			},
			{ transaction },
		);

		// 写入积分流水
		const log = await this.ctx.model.PointsLog.create(
			{
				userId,
				type: 1,
				source,
				points,
				balance: newPoints,
				growthDelta,
				remark,
				expireAt: new Date(Date.now() + 365 * 86400000),
			},
			{ transaction },
		);

		// 写入积分批次（FIFO）
		await this.ctx.model.PointsBatch.create(
			{
				userId,
				source,
				original: points,
				remaining: points,
				expireAt: new Date(Date.now() + 365 * 86400000),
				logId: (log as any).id,
			},
			{ transaction },
		);

		// 检查等级升级
		const levelUpResult = await this.ctx.service.member.checkAndUpgrade(
			userId,
			newGrowthValue,
			transaction,
		);

		return {
			currentPoints: newPoints,
			currentGrowth: newGrowthValue,
			isLevelUp: levelUpResult.upgraded,
		};
	}
```

```typescript
	/**
	 * 处理单个任务的进度更新
	 */
	private async processTask(
		userId: number,
		task: any,
		eventData: Record<string, any>,
	): Promise<void> {
		// 1. 条件匹配
		if (!this.matchCondition(task.condition, eventData)) return;

		// 2. 等级门槛检查
		if (task.required_level) {
			const member = await this.ctx.model.UserMember.findOne({
				where: { userId },
				attributes: ['levelCode'],
				raw: true,
			});
			if (!member) return;
			const levelOrder = ['free', 'silver', 'gold', 'diamond', 'black'];
			const userLevelIdx = levelOrder.indexOf((member as any).levelCode);
			const requiredIdx = levelOrder.indexOf(task.required_level);
			if (userLevelIdx < requiredIdx) return;
		}

		// 3. 获取/创建用户任务进度
		const cycleKey = this.getCycleKey(task.reset_cycle);
		const [userTask, created] = await this.ctx.model.UserTask.findOrCreate({
			where: { userId, taskCode: task.code, cycleKey },
			defaults: { progress: 0, status: 'pending' },
		});

		const ut = userTask as any;

		// 4. 如果已完成或已领取，跳过
		if (ut.status === 'completed' || ut.status === 'claimed') return;

		// 5. 更新进度
		const newProgress = ut.progress + 1;
		const updates: any = { progress: newProgress };

		if (newProgress >= task.progress_target) {
			updates.status = 'completed';
			updates.completedAt = new Date();
		}

		await ut.update(updates);
	}

	/**
	 * 条件匹配
	 */
	private matchCondition(
		condition: any,
		eventData: Record<string, any>,
	): boolean {
		if (!condition) return true;

		// 解析 JSON condition
		const cond =
			typeof condition === 'string' ? JSON.parse(condition) : condition;

		for (const [key, value] of Object.entries(cond)) {
			// 特殊字段：distinct_field 表示需要去重计数（由调用方保证）
			if (key === 'distinct_field') continue;

			// streak 条件：检查事件数据中的 streak 是否达标
			if (key === 'streak') {
				if (!eventData.streak || eventData.streak < (value as number))
					return false;
				continue;
			}

			// amount 条件：检查累计金额
			if (key === 'amount') {
				if (!eventData.amount || eventData.amount < (value as number))
					return false;
				continue;
			}

			// login_days 条件
			if (key === 'login_days') {
				if (!eventData.loginDays || eventData.loginDays < (value as number))
					return false;
				continue;
			}

			// 普通字段精确匹配
			if (eventData[key] !== value) return false;
		}

		return true;
	}

	/**
	 * 生成周期键
	 */
	private getCycleKey(resetCycle: string): string {
		const now = new Date();
		switch (resetCycle) {
			case 'daily':
				return now.toISOString().slice(0, 10); // 2026-05-26
			case 'weekly': {
				// ISO 周数
				const d = new Date(now);
				d.setHours(0, 0, 0, 0);
				d.setDate(d.getDate() + 3 - ((d.getDay() + 6) % 7));
				const week1 = new Date(d.getFullYear(), 0, 4);
				const weekNum =
					1 +
					Math.round(
						((d.getTime() - week1.getTime()) / 86400000 -
							3 +
							((week1.getDay() + 6) % 7)) /
							7,
					);
				return `${d.getFullYear()}-W${String(weekNum).padStart(2, '0')}`;
			}
			case 'monthly':
				return now.toISOString().slice(0, 7); // 2026-05
			case 'yearly':
				return String(now.getFullYear()); // 2026
			case 'once':
			default:
				return 'once';
		}
	}

	/**
	 * 检查每日上限
	 */
	private async checkDailyCap(
		userId: number,
		capGroup: string,
		points: number,
	): Promise<boolean> {
		const today = new Date().toISOString().slice(0, 10);
		const cap = await this.ctx.model.DailyPointsCap?.findOne?.({
			where: { userId, capDate: today, capGroup },
			raw: true,
		});

		} catch {

		}

		const currentEarned = (cap as any)?.earned || 0;
		return currentEarned + points <= maxPoints;
	}

	/**
```
```typescript
	 * 自动履约
	 */
	private async fulfill(
		userId: number,
		order: any,
	): Promise<{ status: string; data: any }> {
		const config = order.fulfillConfig;
		let result: any = null;
		let status = 'fulfilled';

		try {
			switch (order.category || this.getCategoryFromConfig(config)) {
				case 'member_days':
					result = await this.fulfillMemberDays(userId, config);
					break;
				case 'coupon':
					result = await this.fulfillCoupon(userId, config);
					break;
				case 'tool_unlock':
					result = await this.fulfillToolUnlock(userId, config);
					break;
				case 'badge':
					result = await this.fulfillBadge(userId, config);
					break;
				default:
					status = 'failed';
					result = { error: '未知商品类型' };
			}
		} catch (err: any) {
			status = 'failed';
			result = { error: err.message };
		}

		// 更新订单履约状态
		await this.ctx.model.PointsMallOrder.update(
			{
				fulfillStatus: status,
				fulfillResult: result,
				fulfilledAt: status === 'fulfilled' ? new Date() : null,
			},
			{ where: { orderNo: order.orderNo } },
		);

		return { status, data: result };
	}
```

```typescript
// app/service/pointsExpire.ts
import BaseService from './base';

/**
 * 积分过期服务
 *
 * 规则：
 * - 积分获得日 + 365 天过期
 * - 消耗时 FIFO（优先扣除最早获得的批次）
 * - 每日凌晨扫描过期批次，清零并写入流水
 * - T-30/T-7/T-0 三级提醒
 */
export default class PointsExpireService extends BaseService {
	/**
	 * 执行过期清零（定时任务调用）
	 * 扫描所有 expire_at <= NOW 且 remaining > 0 且 is_expired = 0 的批次
	 */
	async processExpiredBatches(): Promise<{
		processedUsers: number;
		totalExpired: number;
	}> {
		const { Op } = require('sequelize');
		const now = new Date();

		// 分批处理，每次最多 500 条
		let totalExpired = 0;
		const processedUserIds = new Set<number>();

		while (true) {
			const batches = await this.ctx.model.PointsBatch.findAll({
				where: {
					isExpired: 0,
					remaining: { [Op.gt]: 0 },
					expireAt: { [Op.lte]: now },
				},
				limit: 500,
				order: [['expire_at', 'ASC']],
			});

			if (batches.length === 0) break;

			for (const batch of batches) {
				const b = batch as any;
				const expiredPoints = b.remaining;

				await this.ctx.model.transaction(async (t: any) => {
					// 标记批次过期
					await b.update({ remaining: 0, isExpired: 1 }, { transaction: t });

					// 扣减用户积分余额
					const member = await this.ctx.model.UserMember.findOne({
						where: { userId: b.userId },
						lock: t.LOCK.UPDATE,
						transaction: t,
					});
					if (member) {
						const newPoints = Math.max(
							0,
							(member as any).points - expiredPoints,
						);
						await (member as any).update(
							{ points: newPoints },
							{ transaction: t },
						);

						// 写入过期流水
						await this.ctx.model.PointsLog.create(
							{
								userId: b.userId,
								type: 3, // 过期
								source: 'points_expire',
								points: -expiredPoints,
								balance: newPoints,
								growthDelta: 0,
								remark: `积分过期清零（批次#${b.id}）`,
							},
							{ transaction: t },
						);
					}
				});

				totalExpired += expiredPoints;
				processedUserIds.add(b.userId);
			}
		}

		// 发送过期通知
		for (const userId of processedUserIds) {
			try {
				await (this.ctx.service.notification as any).core.send({
					typeCode: 'BUSINESS_POINTS_EXPIRE',
					userId,
					variables: { expiredPoints: totalExpired },
				});
			} catch {
				/* ignore */
			}
		}

		return { processedUsers: processedUserIds.size, totalExpired };
	}

	/**
	 * 发送过期提醒（T-30 / T-7 / T-0）
	 */
	async sendExpireReminders(): Promise<void> {
		const { Op } = require('sequelize');
		const remindDays = [30, 7, 0];

		for (const days of remindDays) {
			const targetDate = new Date(Date.now() + days * 86400000);
			const startOfDay = new Date(
				targetDate.toISOString().slice(0, 10) + 'T00:00:00',
			);
			const endOfDay = new Date(
				targetDate.toISOString().slice(0, 10) + 'T23:59:59',
			);

			// 查找即将过期的批次（按用户聚合）
			const { fn, col } = require('sequelize');
			const results = await this.ctx.model.PointsBatch.findAll({
				attributes: ['userId', [fn('SUM', col('remaining')), 'totalExpiring']],
				where: {
					isExpired: 0,
					remaining: { [Op.gt]: 0 },
					expireAt: { [Op.between]: [startOfDay, endOfDay] },
				},
				group: ['user_id'],
				raw: true,
			});

			for (const row of results as any[]) {
				if (row.totalExpiring <= 0) continue;
				try {
					await (this.ctx.service.notification as any).core.send({
						typeCode: 'BUSINESS_POINTS_EXPIRE_REMIND',
						userId: row.userId,
						variables: {
							expirePoints: row.totalExpiring,
							expireDays: days,
							expireDate: targetDate.toISOString().slice(0, 10),
						},
					});
				} catch {
					/* ignore */
				}
			}
		}
	}

	/**
	 * FIFO 消耗积分（从最早的批次开始扣）
	 * 在 member.consumePoints 时调用
	 */
	async consumeFIFO(
		userId: number,
		amount: number,
		transaction: any,
	): Promise<void> {
		const { Op } = require('sequelize');
		let remaining = amount;

		const batches = await this.ctx.model.PointsBatch.findAll({
			where: {
				userId,
				isExpired: 0,
				remaining: { [Op.gt]: 0 },
			},
			order: [['expire_at', 'ASC']], // FIFO：最早过期的先扣
			lock: transaction.LOCK.UPDATE,
			transaction,
		});

		for (const batch of batches) {
			if (remaining <= 0) break;
			const b = batch as any;
			const deduct = Math.min(remaining, b.remaining);
			await b.update({ remaining: b.remaining - deduct }, { transaction });
			remaining -= deduct;
		}
	}

	/**
	 * 管理端：过期统计
	 */
	async getExpireStats() {
		const { Op, fn, col } = require('sequelize');
		const now = new Date();

		// 即将过期（30天内）
		const soonExpire = await this.ctx.model.PointsBatch.findOne({
			attributes: [[fn('SUM', col('remaining')), 'total']],
			where: {
				isExpired: 0,
				remaining: { [Op.gt]: 0 },
				expireAt: {
					[Op.between]: [now, new Date(Date.now() + 30 * 86400000)],
				},
			},
			raw: true,
		});

		// 已过期总量（本月）
		const monthStart = new Date(now.toISOString().slice(0, 7) + '-01');
		const expiredThisMonth = await this.ctx.model.PointsLog.findOne({
			attributes: [[fn('SUM', fn('ABS', col('points'))), 'total']],
			where: {
				type: 3,
				createdAt: { [Op.gte]: monthStart },
			},
			raw: true,
		});

		return {
			soonExpirePoints: Number((soonExpire as any)?.total) || 0,
			expiredThisMonth: Number((expiredThisMonth as any)?.total) || 0,
		};
	}
}
```
