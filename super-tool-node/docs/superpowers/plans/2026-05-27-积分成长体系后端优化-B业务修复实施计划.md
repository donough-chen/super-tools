- [ ] **Step 3: 修改 member.refundPoints**

打开 `app/service/member.ts`，找到 `refundPoints` 方法，整体重构：

```typescript
  /**
   * 退款回扣积分（v2 + spec §2.7-#21 反向 FIFO 改造）
   *
   * 规则：
   *   - 不扣成长值（growth_delta=0）
   *   - 允许负余额（user_members.points 已 ALTER 为 SIGNED；spec A1）
   *   - 优先扣回"原批次"的 points_remaining
   *   - feature flag mall.refund.reverse_fifo=true 时：原批次不够，反向 FIFO 恢复已消费批次
   *   - 仍不够才扣会员余额
   */
  async refundPoints(
    userId: number,
    originalLogId: number,
    refundAmount: number,
    options: { remark?: string; transaction?: any } = {},
  ) {
    if (refundAmount <= 0) this.ctx.throw(400, '退款积分必须为正数');

    const ownTx = !options.transaction;
    const t: any = options.transaction || await (this.ctx.model as any).transaction();
    const { Op } = require('sequelize');

    try {
      const original: any = await this.ctx.model.PointsLog.findByPk(originalLogId, {
        lock: t.LOCK.UPDATE, transaction: t,
      });
      if (!original) this.ctx.throw(404, '原积分流水不存在');
      if (original.userId !== userId) this.ctx.throw(403, '原流水不属于该用户');

      const member: any = await this.ctx.model.UserMember.findOne({
        where: { userId }, lock: t.LOCK.UPDATE, transaction: t,
      });
      if (!member) this.ctx.throw(404, '会员记录不存在');

      let toRecover = refundAmount;

      // ===== 1) 优先回写原批次 =====
      if (original.status === 1 && original.pointsRemaining > 0) {
        const recoverFromBatch = Math.min(toRecover, original.pointsRemaining + 0);
        // 注：原批次 pointsRemaining 是"还没扣的额度"，可以加回（不能超过 |original.points|）
        const newBatchRemaining = original.pointsRemaining - 0 + recoverFromBatch;
        // 上限：不能超过原入账总额
        const maxRemaining = Math.abs(original.points);
        const finalRemaining = Math.min(newBatchRemaining, maxRemaining);
        const actuallyAdded = finalRemaining - original.pointsRemaining;
        await original.update(
          { pointsRemaining: finalRemaining, status: 1 },
          { transaction: t },
        );
        toRecover -= actuallyAdded;
      } else if (original.status === 2 || original.status === 1) {
        // 原批次已耗尽（status=2 或 status=1 且 pointsRemaining=0）
        // → 把它"复活"为 status=1
        const maxRemaining = Math.abs(original.points);
        const recoverFromBatch = Math.min(toRecover, maxRemaining);
        await original.update(
          { pointsRemaining: original.pointsRemaining + recoverFromBatch, status: 1 },
          { transaction: t },
        );
        toRecover -= recoverFromBatch;
      }

      // ===== 2) 反向 FIFO（feature flag）=====
      const reverseFifoEnabled = await this.getReverseFifoFlag();
      if (reverseFifoEnabled && toRecover > 0) {
        // 找最近消费过的批次（status=2 + 同用户 + type=1 即正向积分批次），按 id DESC
        const consumedBatches: any[] = await this.ctx.model.PointsLog.findAll({
          where: {
            userId, type: 1, status: 2,
            id: { [Op.ne]: originalLogId },  // 排除已处理的原批次
          },
          order: [['id', 'DESC']],
          lock: t.LOCK.UPDATE, transaction: t,
        });
        for (const b of consumedBatches) {
          if (toRecover <= 0) break;
          const maxRecoverable = Math.abs(b.points);  // 该批次原入账总额
          const recoverHere = Math.min(toRecover, maxRecoverable - b.pointsRemaining);
          if (recoverHere <= 0) continue;
          await b.update(
            {
              pointsRemaining: b.pointsRemaining + recoverHere,
              status: 1,  // 复活
            },
            { transaction: t },
          );
          toRecover -= recoverHere;
        }
      }

      // ===== 3) 余下的扣 / 加 user_members.points（已 SIGNED 支持负值）=====
      // 实际可用余额 += refundAmount（用户拿回的实际可用积分）
      // 注：在 step 1/2 中已经把"恢复批次"加到了 points_remaining；
      // 但 user_members.points 是"批次余额之和"，所以这里也要 + refundAmount
      const newBalance = member.points + refundAmount;
      await member.update({ points: newBalance }, { transaction: t });

      // ===== 4) 写一条 type=2 source='refund' 的流水（balance 已 SIGNED）=====
      const log: any = await this.ctx.model.PointsLog.create(
        {
          userId,
          type: 2,                 // 消费类（这里是退款入账，但语义上"恢复消费记账"，仍保持 type=2）
          source: 'refund',
          points: refundAmount,    // 正数：退回积分
          balance: newBalance,
          growthDelta: 0,
          bizType: 'refund',
          bizId: String(originalLogId),
          remark: options.remark || `退款回扣原批次 #${originalLogId}（反向 FIFO ${reverseFifoEnabled ? 'on' : 'off'}）`,
          pointsRemaining: 0,
          status: 2,
          sourceLevelId: member.levelId,
          sourceEvent: 'refund',
          growthMultiplier: original.growthMultiplier || 1.0,
        },
        { transaction: t },
      );

      if (ownTx) await t.commit();
      return { logId: log.id, balance: newBalance };
    } catch (err) {
      if (ownTx) await t.rollback();
      throw err;
    }
  }

  /** 读取反向 FIFO feature flag（system_configs.mall.refund.reverse_fifo=true 时启用） */
  private async getReverseFifoFlag(): Promise<boolean> {
    try {
      const rows: any[] = await (this.app.model as any).query(
        "SELECT `value` FROM `system_configs` WHERE `group`='mall' AND `key`='refund.reverse_fifo' LIMIT 1",
        { type: (this.app.model as any).QueryTypes.SELECT },
      );
      if (rows.length > 0) return String(rows[0].value).toLowerCase() === 'true';
    } catch { /* ignore */ }
    return false;  // 默认关闭
  }
```

```typescript
/**
 * 字段白名单提取（admin 接口防字段注入）
 *  设计依据: docs/superpowers/specs/2026-05-27-积分成长体系后端优化设计文档.md §2.10
 */
export function pickFields<T extends Record<string, any>>(
	source: any,
	fields: readonly string[],
): Partial<T> {
	if (!source || typeof source !== 'object') return {};
	const result: any = {};
	for (const f of fields) {
		if (Object.prototype.hasOwnProperty.call(source, f)) {
			result[f] = source[f];
		}
	}
	return result;
}
```

- [ ] **Step 1: 写脚本骨架**

创建 `scripts/migrate_existing_users_points.ts`：

```typescript
#!/usr/bin/env node
/**
 * 历史用户积分体系迁移脚本（一次性）
 *  设计依据: docs/superpowers/specs/2026-05-27-积分成长体系后端优化设计文档.md §6
 *
 *  目的：
 *    1. 按 growth_value 重新计算 user_members.levelId
 *    2. 现有 type=1 status=1 的 points_logs 补 expire_at（按当前等级 pointsExpireDays）
 *    3. 输出迁移报告 CSV
 *
 *  幂等：可重跑，已正确等级 / 已有 expire_at 跳过
 *
 *  使用：
 *    cd super-tool-node
 *    npx ts-node scripts/migrate_existing_users_points.ts --dry-run
 *    npx ts-node scripts/migrate_existing_users_points.ts --apply
 */
import { Application } from 'egg';
import * as fs from 'fs';

const args = process.argv.slice(2);
const DRY_RUN = !args.includes('--apply');

async function main() {
	const egg = require('egg');
	const app: Application = await egg.start({
		baseDir: process.cwd(),
		workers: 1,
	} as any);

	console.log(`[migrate] mode = ${DRY_RUN ? 'DRY-RUN' : 'APPLY'}`);

	const failures: any[] = [];
	let updated = 0;
	let skipped = 0;

	// 1) 分批读 user_members（1000 条一批）
	const BATCH = 1000;
	let offset = 0;
	while (true) {
		const members: any[] = await (app.model as any).UserMember.findAll({
			attributes: ['userId', 'levelId', 'growthValue'],
			order: [['user_id', 'ASC']],
			limit: BATCH,
			offset,
		});
		if (members.length === 0) break;

		for (const m of members) {
			try {
				await (app.model as any).transaction(async (t: any) => {
					// 计算正确 levelId
					const levels: any[] = await (app.model as any).MemberLevel.findAll({
						order: [['level', 'ASC']],
					});
					let correctLevelId = levels[0].id;
					for (const lv of levels) {
						if (m.growthValue >= (lv.growthThreshold || 0)) {
							correctLevelId = lv.id;
						}
					}

					if (correctLevelId === m.levelId) {
						skipped++;
						return;
					}

					// 拿到该等级的 pointsExpireDays
					const targetLevel = levels.find(l => l.id === correctLevelId);
					const benefits = targetLevel?.benefits || {};
					const expireDays = Number(benefits.points_expire_days) || 365;

					if (!DRY_RUN) {
						// 1) 更新 levelId
						await (app.model as any).UserMember.update(
							{ levelId: correctLevelId, levelCode: targetLevel.code },
							{ where: { userId: m.userId }, transaction: t },
						);

						// 2) 补 expire_at（仅对 expire_at IS NULL 的存量批次）
						const newExpireAt = new Date(Date.now() + expireDays * 86_400_000);
						const { Op } = require('sequelize');
						await (app.model as any).PointsLog.update(
							{ expireAt: newExpireAt },
							{
								where: {
									userId: m.userId,
									type: 1,
									status: 1,
									pointsRemaining: { [Op.gt]: 0 },
									expireAt: null,
								},
								transaction: t,
							},
						);
					}

					updated++;
					console.log(
						`[migrate] user=${m.userId} levelId ${m.levelId} → ${correctLevelId}`,
					);
				});
			} catch (err: any) {
				failures.push({ userId: m.userId, reason: err.message });
				console.error(`[migrate] user=${m.userId} FAIL: ${err.message}`);
			}
		}

		offset += BATCH;
		if (members.length < BATCH) break;
	}

	// 2) 输出报告
	const reportPath = `migration_report_${Date.now()}.csv`;
	const lines = ['userId,reason'];
	for (const f of failures)
		lines.push(`${f.userId},${f.reason.replace(/,/g, ';')}`);
	fs.writeFileSync(reportPath, lines.join('\n'));

	console.log(`\n=========== Migration Summary ===========`);
	console.log(`Mode:    ${DRY_RUN ? 'DRY-RUN' : 'APPLY'}`);
	console.log(`Updated: ${updated}`);
	console.log(`Skipped: ${skipped}`);
	console.log(`Failed:  ${failures.length} (see ${reportPath})`);

	await app.close();
}

main().catch(err => {
	console.error(err);
	process.exit(1);
});
```
