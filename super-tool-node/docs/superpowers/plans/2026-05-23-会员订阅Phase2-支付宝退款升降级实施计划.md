```typescript
import { AlipayProvider } from './alipay';

export function createProvider(name: ProviderCode, ctx: any): PaymentProvider {
  if (name === 'mock') return new MockProvider();
  if (name === 'alipay') {
    const cfg = ctx.app.config.alipay || {};
    return new AlipayProvider({
      appId: cfg.appId,
      privateKey: cfg.merchantPrivateKey,
      alipayPublicKey: cfg.publicKey,
      gateway: cfg.gateway,
      signType: cfg.signType,
      notifyUrl: cfg.notifyUrl,
      returnUrl: cfg.returnUrl,
    });
  }
  if (name === 'wechat_jsapi') throw new Error('wechat_jsapi 暂未接入');
  throw new Error(`Unknown provider: ${name}`);
}
```

```typescript
import { calcSwitchPlan } from '../../../app/lib/payment/priceCalculator';

const monthlyPlan = { code: 'monthly', price: 6.8, durationDays: 30 };
const yearlyPlan = { code: 'yearly', price: 68, durationDays: 365 };
const liftimePlan = { code: 'lifetime', price: 999, durationDays: 0 };
const NOW = new Date('2026-05-23T00:00:00Z');

describe('calcSwitchPlan', () => {
  it('未付费用户 → scene=1 新购', () => {
    const r = calcSwitchPlan({
      currentMember: { paidPlanCode: null, paidExpireAt: null },
      newPlan: monthlyPlan, now: NOW,
    });
    expect(r.scene).toBe(1);
    expect(r.amount).toBe(6.8);
    expect(r.remainingValue).toBe(0);
  });

  it('付费已过期 → scene=1 新购', () => {
    const r = calcSwitchPlan({
      currentMember: { paidPlanCode: 'monthly', paidExpireAt: '2026-05-01T00:00:00Z' },
      newPlan: monthlyPlan, now: NOW,
    });
    expect(r.scene).toBe(1);
  });

  it('付费同套餐 → scene=2 续费', () => {
    const r = calcSwitchPlan({
      currentMember: { paidPlanCode: 'monthly', paidExpireAt: '2026-06-22T00:00:00Z' }, // 还剩30天
      currentPlan: monthlyPlan,
      newPlan: monthlyPlan, now: NOW,
    });
    expect(r.scene).toBe(2);
    expect(r.amount).toBe(6.8);
    expect(r.remainingDays).toBe(30);
  });

  it('付费 monthly → 升级 yearly → scene=3 + 差价计算正确', () => {
    // monthly 还剩 25 天 → remainingValue = 6.8 * 25/30 ≈ 5.67
    // 差价 = 68 - 5.67 = 62.33
    const expireAt = new Date(NOW.getTime() + 25 * 86400000);
    const r = calcSwitchPlan({
      currentMember: { paidPlanCode: 'monthly', paidExpireAt: expireAt },
      currentPlan: monthlyPlan,
      newPlan: yearlyPlan, now: NOW,
    });
    expect(r.scene).toBe(3);
    expect(r.remainingDays).toBe(25);
    expect(r.remainingValue).toBeCloseTo(5.67, 1);
    expect(r.amount).toBeCloseTo(62.33, 1);
  });

  it('付费 yearly → 降级 monthly → scene=4 + amount=0 + 折算天数', () => {
    // yearly 还剩 200 天 → remainingValue = 68 * 200/365 ≈ 37.26
    // 折算天数 = 37.26 / 6.8 * 30 = 164 天
    const expireAt = new Date(NOW.getTime() + 200 * 86400000);
    const r = calcSwitchPlan({
      currentMember: { paidPlanCode: 'yearly', paidExpireAt: expireAt },
      currentPlan: yearlyPlan,
      newPlan: monthlyPlan, now: NOW,
    });
    expect(r.scene).toBe(4);
    expect(r.amount).toBe(0);
    const daysFromValue = Math.floor((r.newExpireAt.getTime() - NOW.getTime()) / 86400000);
    expect(daysFromValue).toBeGreaterThanOrEqual(160);
    expect(daysFromValue).toBeLessThanOrEqual(170);
  });

  it('永久会员升降级抛错', () => {
    expect(() => calcSwitchPlan({
      currentMember: { paidPlanCode: 'lifetime', paidExpireAt: new Date('2099-12-31') },
      currentPlan: liftimePlan,
      newPlan: monthlyPlan, now: NOW,
    })).toThrow(/永久/);
  });

  it('remainingDays=0 边界（恰好今天到期）', () => {
    const r = calcSwitchPlan({
      currentMember: { paidPlanCode: 'monthly', paidExpireAt: new Date(NOW.getTime() + 1000) },
      currentPlan: monthlyPlan,
      newPlan: yearlyPlan, now: NOW,
    });
    expect(r.scene).toBe(3);
    expect(r.remainingValue).toBeCloseTo(0, 1);
    expect(r.amount).toBeCloseTo(68, 1);
  });

  it('newExpireAt 升级以 NOW 为基准', () => {

      currentPlan: monthlyPlan,
      newPlan: yearlyPlan, now: NOW,
    });
    const daysFromNow = Math.round((r.newExpireAt.getTime() - NOW.getTime()) / 86400000);
    expect(daysFromNow).toBe(365);
  });
});
```

```typescript
async preview(input: { userId: number; planCode: string }) {
  const { userId, planCode } = input;
  const newPlan = await this.ctx.model.PaidPlan.findOne({ where: { code: planCode, status: 1 }});
  if (!newPlan) this.ctx.throw(404, '套餐不存在或已下架');
  const newPlanData = (newPlan as any).toJSON();

  const member = await this.ctx.model.UserMember.findOne({ where: { userId } });
  const memberData = member ? (member as any).toJSON() : { paidPlanCode: null, paidExpireAt: null };

  let currentPlan;
  let currentPlanName;
  if (memberData.paidPlanCode && memberData.paidExpireAt && new Date(memberData.paidExpireAt) > new Date()) {
    const cp = await this.ctx.model.PaidPlan.findOne({ where: { code: memberData.paidPlanCode } });
    if (cp) {
      const d = (cp as any).toJSON();
      currentPlan = { code: d.code, price: Number(d.price), durationDays: d.durationDays };
      currentPlanName = d.name;
    }
  }

  const calc = calcSwitchPlan({
    currentMember: memberData,
    currentPlan,
    newPlan: { code: newPlanData.code, price: Number(newPlanData.price), durationDays: newPlanData.durationDays },
  });

  return {
    scene: calc.scene,
    amount: calc.amount.toFixed(2),
    remainingDays: calc.remainingDays,
    remainingValue: calc.remainingValue.toFixed(2),
```

# 支付宝沙箱接入指南

## 1. 申请沙箱账号

访问 [Alipay Open Platform 沙箱](https://open.alipay.com/develop/sandbox/app)：

- 选 RSA2 + 2048

### 2.3 上传应用公钥到沙箱
- 沙箱后台 → 开发设置 → 接口加签方式 → 公钥

- 上一步保存后，沙箱后台显示"支付宝公钥"（与 AppId 绑定）

ALIPAY_APP_ID=2021000000000000              # 沙箱 AppId
ALIPAY_MERCHANT_PRIVATE_KEY="-----BEGIN RSA PRIVATE KEY-----\n...\n-----END RSA PRIVATE KEY-----"

ALIPAY_GATEWAY=https://openapi.alipaydev.com/gateway.do
ALIPAY_RETURN_URL=http://localhost:7001/api/payments/alipay/return
# 可选：内网穿透后填，否则留空仅靠主动 query

- 买家账号 / 登录密码 / 支付密码

- 启动隧道：`cpolar http 7001`

## 风险与回滚

### 关键风险

| 风险 | 触发条件 | 缓解 |
|---|---|---|
| alipay-sdk v3 API 变更 | npm install 拉到不兼容版本 | package.json 锁 `^4.x` 已知稳定版 |
| 异步通知签名 fail | 公钥配错 | T9 dev 启动后单测验证；spec § 11 文档详述区分 |
| 主动 query 与 webhook 双发重复处理 | T7 实装漏掉幂等 | _applyPaymentSuccess 行级锁 + status===1 直接 return |
| OrderService.create 重构破坏 phase1 | T5 改动错误 | T0 跑 phase1 smoke 基线；T5 完成后再跑一次 |
| Refund 事务嵌套问题 | RefundService 内调 provider.refund 时间长 | provider.refund 在 transaction 内同步等；超时由 alipay-sdk 10s |
| H5 升降级 modal 误导用户 | 文案不清 | 二次确认 + 显示新到期时间 |
| .env.local 私钥泄漏 | 不小心 git add 了 | .gitignore 已含；推荐 husky pre-commit 扫密钥（可作为 follow-up） |

### 回滚策略

- **半路撤回**：分支内 `git reset --hard <last-good-commit>` 重做
- **整体撤回**：merge 后 `git revert -m 1 <merge-commit>`
- **DB 回滚 SQL**（保留草稿）：

```sql
DROP TABLE IF EXISTS member_refunds;
ALTER TABLE member_orders DROP COLUMN source_plan_code;
ALTER TABLE member_orders DROP COLUMN source_remaining_value;
DELETE FROM permissions WHERE code IN ('member:refund', 'member:refund:create', 'system:dev:trigger-schedule');
DELETE FROM notification_types WHERE code = 'BUSINESS_PAYMENT_REFUNDED';
DELETE FROM system_configs WHERE `group`='payment' AND `key` LIKE 'alipay%';
```
