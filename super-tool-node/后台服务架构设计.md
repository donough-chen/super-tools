## 一、技术栈调研

### 1.1 核心技术选型分析

#### Node.js
| 维度 | 说明 |
|------|------|
| 版本推荐 | Node.js 18.x LTS / 20.x LTS |
| 优势 | 非阻塞I/O、事件驱动、高并发处理能力 |
| 适用场景 | API服务、实时通信、微服务 |
| 性能指标 | 单机可支撑 10k+ QPS |

#### Egg.js
| 维度 | 说明 |
|------|------|
| 版本推荐 | Egg.js 3.x |
| 核心特性 | 约定优于配置、插件机制、多进程模型 |
| 优势 | 企业级框架、内置安全机制、完善的生态 |
| 对比Koa | 在Koa基础上封装，提供更完整的企业级解决方案 |

#### TypeScript
| 维度 | 说明 |
|------|------|
| 版本推荐 | TypeScript 5.x |
| 优势 | 类型安全、IDE支持、代码可维护性 |
| 编译工具 | ts-node / tsc |

#### MySQL
| 维度 | 说明 |
|------|------|
| 版本推荐 | MySQL 8.0+ |
| ORM选择 | Sequelize 6.x（兼容现有 egg-mysql 的多数据源模式） |
| 连接池 | egg-sequelize 内置连接池管理 |
| 现有数据源 | defaultDB (yoyo主库)、dnfCmsDB (DNF管理端)、xinyueDB (心悦) |
| 迁移注意 | 现有 30+ 张表需转化为 Sequelize Model，保持字段兼容 |

#### Redis
| 维度 | 说明 |
|------|------|
| 版本推荐 | Redis 7.x |
| 用途 | 缓存、Session、消息队列、分布式锁 |
| 客户端 | ioredis |

### 4.2 Sequelize 模型定义

```typescript
// app/model/user.ts
import { Application } from 'egg';
import {
  DataTypes,
  Model,
  Optional,
  Association,
} from 'sequelize';

// 用户属性接口
export interface UserAttributes {
  id: number;
  uuid: string;
  username: string;
  email: string;
  phone?: string;
  password: string;
  avatar?: string;
  nickname?: string;
  status: 0 | 1;
  lastLogin?: Date;
  loginIp?: string;
  createdAt?: Date;
  updatedAt?: Date;
  deletedAt?: Date;
}

// 创建时可选字段
export interface UserCreationAttributes
  extends Optional<UserAttributes, 'id' | 'uuid' | 'status'> {}

export class UserModel
  extends Model<UserAttributes, UserCreationAttributes>
  implements UserAttributes
{
  public id!: number;
  public uuid!: string;
  public username!: string;
  public email!: string;
  public phone?: string;
  public password!: string;
  public avatar?: string;
  public nickname?: string;
  public status!: 0 | 1;
  public lastLogin?: Date;
  public loginIp?: string;
  public readonly createdAt!: Date;
  public readonly updatedAt!: Date;
  public readonly deletedAt?: Date;

  // 关联声明
  public static associations: {
    roles: Association<UserModel, any>;
  };
}

export default (app: Application) => {
  const { STRING, INTEGER, TINYINT, DATE, UUID, UUIDV4 } = app.Sequelize;

  UserModel.init(
    {
      id: {
        type: INTEGER.UNSIGNED,
        autoIncrement: true,
        primaryKey: true,
      },
      uuid: {
        type: UUID,
        defaultValue: UUIDV4,
        allowNull: false,
        unique: true,
      },
      username: {
        type: STRING(50),
        allowNull: false,
        unique: true,
        validate: {
          len: [3, 50],
          isAlphanumeric: true,
        },
      },
      email: {
        type: STRING(100),
        allowNull: false,
        unique: true,
        validate: {
          isEmail: true,
        },
      },
      phone: {
        type: STRING(20),
        allowNull: true,
      },
      password: {
        type: STRING(255),
        allowNull: false,
      },
      avatar: {
        type: STRING(500),
        allowNull: true,
      },
      nickname: {
        type: STRING(50),
        allowNull: true,
      },
      status: {
        type: TINYINT,
        defaultValue: 1,
        validate: {
          isIn: [[0, 1]],
        },
      },
      lastLogin: {
        type: DATE,
        allowNull: true,
      },
      loginIp: {
        type: STRING(50),
        allowNull: true,
      },
    },
    {
      sequelize: app.sequelize,
      tableName: 'users',
      paranoid: true,
      timestamps: true,
      underscored: true,
    }
  );

  // 定义关联关系
  (UserModel as any).associate = () => {
    UserModel.belongsToMany(app.model.Role, {
      through: 'user_roles',
      foreignKey: 'user_id',
      otherKey: 'role_id',
      as: 'roles',
    });
  };

  return UserModel;
};
```

### 5.1 错误处理中间件

```typescript
// app/middleware/errorHandler.ts
import { Context, Application } from 'egg';

export default (_options: any, _app: Application) => {
  return async (ctx: Context, next: () => Promise<void>) => {
    try {
      await next();
    } catch (err: any) {
      // 记录错误日志
      ctx.logger.error('[ErrorHandler]', err);

      const status = err.status || 500;
      const message = err.message || 'Internal Server Error';

      // 区分不同错误类型
      if (status === 422) {
        // 参数验证错误
        ctx.body = {
          code: 422,
          message: '参数验证失败',
          errors: err.errors,
        };
      } else if (status === 401) {
        ctx.body = {
          code: 401,
          message: '未授权，请先登录',
        };
      } else if (status === 403) {
        ctx.body = {
          code: 403,
          message: '权限不足',
        };
      } else if (status === 404) {
        ctx.body = {
          code: 404,
          message: '资源不存在',
        };
      } else {
        ctx.body = {
          code: status,
          message: process.env.NODE_ENV === 'production'
            ? 'Internal Server Error'
            : message,
        };
      }

      ctx.status = status;
    }
  };
};
```

```typescript
import { Context, Application } from 'egg';

export default (_options: any, app: Application) => {
  return async (ctx: Context, next: () => Promise<void>) => {
    // 白名单路由跳过认证
    const whiteList = [
      '/api/auth/login',
      '/api/auth/register',
      '/api/auth/refresh',
      '/api/health',
    ];

    if (whiteList.some(path => ctx.path.startsWith(path))) {
      return await next();
    }

    const token = ctx.get('Authorization')?.replace('Bearer ', '');

    if (!token) {
      ctx.throw(401, '请提供认证Token');
    }

    try {
      // 检查Token黑名单
      const isBlacklisted = await app.redis.get(
        `${app.config.appConfig.tokenBlacklistPrefix}${token}`
      );

      if (isBlacklisted) {
        ctx.throw(401, 'Token已失效，请重新登录');
      }

      // 验证Token
      const decoded = app.jwt.verify(token, app.config.jwt.secret) as any;
      ctx.state.user = decoded;
      ctx.state.token = token;

      await next();
    } catch (err: any) {
      if (err.name === 'TokenExpiredError') {
        ctx.throw(401, 'Token已过期，请刷新Token');
      } else if (err.name === 'JsonWebTokenError') {
        ctx.throw(401, 'Token无效');
      }
      throw err;
    }
  };
};
```

```typescript
interface RateLimitOptions {
  max?: number;       // 最大请求次数
  window?: number;    // 时间窗口(秒)
  message?: string;
}

export default (options: RateLimitOptions, app: Application) => {
  const {
    max = 100,
    window = 60,
    message = '请求过于频繁，请稍后再试',
  } = options;

  return async (ctx: Context, next: () => Promise<void>) => {
    const ip = ctx.ip;
    const key = `${app.config.appConfig.rateLimitPrefix}${ip}:${ctx.path}`;

    const current = await app.redis.incr(key);

    if (current === 1) {
      await app.redis.expire(key, window);
    }

    // 设置响应头
    ctx.set('X-RateLimit-Limit', String(max));
    ctx.set('X-RateLimit-Remaining', String(Math.max(0, max - current)));

    if (current > max) {
      ctx.throw(429, message);
    }

    await next();
  };
};
```

```typescript
// app/service/base.ts
import { Service } from 'egg';

export interface PaginationOptions {
  page?: number;
  pageSize?: number;
  order?: [string, 'ASC' | 'DESC'][];
}

export interface PaginationResult<T> {
  list: T[];
  total: number;
  page: number;
  pageSize: number;
  totalPages: number;
}

export default class BaseService extends Service {
  /**
   * 分页查询封装
   */
  protected async paginate<T>(
    model: any,
    options: any,
    pagination: PaginationOptions
  ): Promise<PaginationResult<T>> {
    const {
      page = 1,
      pageSize = this.app.config.appConfig.pageSize,
      order = [['created_at', 'DESC']],
    } = pagination;

    const limit = Math.min(pageSize, this.app.config.appConfig.maxPageSize);
    const offset = (page - 1) * limit;

    const { count, rows } = await model.findAndCountAll({
      ...options,
      limit,
      offset,
      order,
    });

    return {
      list: rows,
      total: count,
      page,
      pageSize: limit,
      totalPages: Math.ceil(count / limit),
    };
  }

  /**
   * Redis缓存封装
   */
  protected async getOrSetCache<T>(
    key: string,
    fetchFn: () => Promise<T>,
    ttl: number = 300
  ): Promise<T> {
    const cached = await this.app.redis.get(key);

    if (cached) {
      return JSON.parse(cached) as T;
    }

    const data = await fetchFn();
    await this.app.redis.setex(key, ttl, JSON.stringify(data));
    return data;
  }

  /**
   * 清除缓存
   */
  protected async clearCache(pattern: string): Promise<void> {
    const keys = await this.app.redis.keys(pattern);
    if (keys.length > 0) {
      await this.app.redis.del(...keys);
    }
  }
}
```

```typescript
// app/service/user.ts
import { v4 as uuidv4 } from 'uuid';
import * as bcrypt from 'bcryptjs';
import BaseService, { PaginationOptions, PaginationResult } from './base';
import { UserAttributes } from '../model/user';

export interface CreateUserDTO {
  username: string;
  email: string;
  password: string;
  phone?: string;
  nickname?: string;
}

export interface UpdateUserDTO {
  nickname?: string;
  phone?: string;
  avatar?: string;
  status?: 0 | 1;
}

export interface UserQueryDTO extends PaginationOptions {
  keyword?: string;
  status?: 0 | 1;
  startDate?: string;
  endDate?: string;
}

export default class UserService extends BaseService {
  private readonly CACHE_PREFIX = 'user:';
  private readonly CACHE_TTL = 600; // 10分钟

  /**
   * 创建用户
   */
  async create(dto: CreateUserDTO): Promise<UserAttributes> {
    const { username, email, password } = dto;

    // 检查用户名/邮箱是否已存在
    const existing = await this.app.model.User.findOne({
      where: {
        [this.app.Sequelize.Op.or]: [{ username }, { email }],
      },
    });

    if (existing) {
      const field = existing.username === username ? '用户名' : '邮箱';
      this.ctx.throw(400, `${field}已被注册`);
    }

    // 密码加密
    const hashedPassword = await bcrypt.hash(password, 12);

    const user = await this.app.model.User.create({
      ...dto,
      uuid: uuidv4(),
      password: hashedPassword,
    });

    // 清除列表缓存
    await this.clearCache(`${this.CACHE_PREFIX}list:*`);

    return user.toJSON() as UserAttributes;
  }

  /**
   * 根据ID查询用户
   */
  async findById(id: number): Promise<UserAttributes | null> {
    const cacheKey = `${this.CACHE_PREFIX}${id}`;

    return this.getOrSetCache(
      cacheKey,
      async () => {
        const user = await this.app.model.User.findByPk(id, {
          attributes: { exclude: ['password'] },
          include: [
            {
              model: this.app.model.Role,
              as: 'roles',
              attributes: ['id', 'name', 'code'],
              through: { attributes: [] },
            },
          ],
        });
        return user?.toJSON() || null;
      },
      this.CACHE_TTL
    );
  }

  /**
   * 分页查询用户列表
   */
  async findList(query: UserQueryDTO): Promise<PaginationResult<UserAttributes>> {
    const { keyword, status, startDate, endDate, ...pagination } = query;
    const { Op } = this.app.Sequelize;

    const where: any = {};

    if (keyword) {
      where[Op.or] = [
        { username: { [Op.like]: `%${keyword}%` } },
        { email: { [Op.like]: `%${keyword}%` } },
        { nickname: { [Op.like]: `%${keyword}%` } },
      ];
    }

    if (status !== undefined) {
      where.status = status;
    }

    if (startDate && endDate) {
      where.createdAt = {
        [Op.between]: [new Date(startDate), new Date(endDate)],
      };
    }

    return this.paginate<UserAttributes>(
      this.app.model.User,
      {
        where,
        attributes: { exclude: ['password'] },
      },
      pagination
    );
  }

  /**
   * 更新用户信息
   */
  async update(id: number, dto: UpdateUserDTO): Promise<UserAttributes> {
    const user = await this.app.model.User.findByPk(id);

    if (!user) {
      this.ctx.throw(404, '用户不存在');
    }

    await user.update(dto);

    // 清除用户缓存
    await this.app.redis.del(`${this.CACHE_PREFIX}${id}`);

    return user.toJSON() as UserAttributes;
  }

  /**
   * 删除用户（软删除）
   */
  async delete(id: number): Promise<void> {
    const user = await this.app.model.User.findByPk(id);

    if (!user) {
      this.ctx.throw(404, '用户不存在');
    }

    await user.destroy(); // 软删除

    // 清除缓存
    await this.app.redis.del(`${this.CACHE_PREFIX}${id}`);
    await this.clearCache(`${this.CACHE_PREFIX}list:*`);
  }

  /**
   * 修改密码
   */
  async changePassword(
    id: number,
    oldPassword: string,
    newPassword: string
  ): Promise<void> {
    const user = await this.app.model.User.findByPk(id);

    if (!user) {
      this.ctx.throw(404, '用户不存在');
    }

    const isValid = await bcrypt.compare(oldPassword, user.password);
    if (!isValid) {
      this.ctx.throw(400, '原密码错误');
    }

    const hashedPassword = await bcrypt.hash(newPassword, 12);
    await user.update({ password: hashedPassword });
  }
}
```

```typescript
// app/service/auth.ts
import * as bcrypt from 'bcryptjs';
import BaseService from './base';

export interface LoginDTO {
  username: string;
  password: string;
}

export interface TokenPair {
  accessToken: string;
  refreshToken: string;
  expiresIn: number;
}

export default class AuthService extends BaseService {
  /**
   * 用户登录
   */
  async login(dto: LoginDTO): Promise<TokenPair> {
    const { username, password } = dto;

    // 查找用户（支持用户名/邮箱登录）
    const { Op } = this.app.Sequelize;
    const user = await this.app.model.User.findOne({
      where: {
        [Op.or]: [{ username }, { email: username }],
        status: 1,
      },
    });

    if (!user) {
      this.ctx.throw(401, '用户名或密码错误');
    }

    // 验证密码
    const isValid = await bcrypt.compare(password, user.password);
    if (!isValid) {
      // 记录失败次数
      await this.recordLoginFailure(username);
      this.ctx.throw(401, '用户名或密码错误');
    }

    // 更新登录信息
    await user.update({
      lastLogin: new Date(),
      loginIp: this.ctx.ip,
    });

    // 生成Token对
    return this.generateTokenPair(user);
  }

  /**
   * 刷新Token
   */
  async refreshToken(refreshToken: string): Promise<TokenPair> {
    try {
      const decoded = this.app.jwt.verify(
        refreshToken,
        this.app.config.jwt.secret
      ) as any;

      if (decoded.type !== 'refresh') {
        this.ctx.throw(401, '无效的刷新Token');
      }

      const user = await this.app.model.User.findByPk(decoded.id);
      if (!user || user.status !== 1) {
        this.ctx.throw(401, '用户不存在或已被禁用');
      }

      // 使旧Token失效
      await this.invalidateToken(refreshToken);

      return this.generateTokenPair(user);
    } catch (err: any) {
      this.ctx.throw(401, 'Token无效或已过期');
    }
  }

  /**
   * 登出
   */
  async logout(token: string): Promise<void> {
    await this.invalidateToken(token);
  }

  /**
   * 生成Token对
   */
  private generateTokenPair(user: any): TokenPair {
    const payload = {
      id: user.id,
      uuid: user.uuid,
      username: user.username,
      email: user.email,
    };

    const accessToken = this.app.jwt.sign(
      { ...payload, type: 'access' },
      this.app.config.jwt.secret,
      { expiresIn: '2h' }
    );

    const refreshToken = this.app.jwt.sign(
      { ...payload, type: 'refresh' },
      this.app.config.jwt.secret,
      { expiresIn: '30d' }
    );

    return {
      accessToken,
      refreshToken,
      expiresIn: 7200, // 2小时
    };
  }

  /**
   * 使Token失效（加入黑名单）
   */
  private async invalidateToken(token: string): Promise<void> {
    const key = `${this.app.config.appConfig.tokenBlacklistPrefix}${token}`;
    await this.app.redis.setex(key, 7 * 24 * 3600, '1'); // 保存7天
  }

  /**
   * 记录登录失败
   */
  private async recordLoginFailure(username: string): Promise<void> {
    const key = `login:fail:${username}`;
    const count = await this.app.redis.incr(key);
    if (count === 1) {
      await this.app.redis.expire(key, 3600); // 1小时内
    }
    if (count >= 5) {
      // 锁定账号逻辑
      this.ctx.logger.warn(`账号 ${username} 登录失败次数过多`);
    }
  }
}
```

```typescript
    const lockKey = `lock:${key}`;
    const lockValue = `${Date.now()}_${Math.random()}`;

    const result = await this.app.redis.set(
      lockKey,
      lockValue,
      'EX',
      ttl,
      'NX'
    );

    return result === 'OK' ? lockValue : null;
  }

  /**
   * 释放分布式锁
   */
  async releaseLock(key: string, lockValue: string): Promise<boolean> {
    const lockKey = `lock:${key}`;
    const script = `
      if redis.call("get", KEYS[1]) == ARGV[1] then
        return redis.call("del", KEYS[1])
      else
        return 0
      end
    `;

    const result = await this.app.redis.eval(script, 1, lockKey, lockValue);
    return result === 1;
  }

  /**
   * 计数器（用于统计、限流）
   */
  async increment(key: string, ttl?: number): Promise<number> {
    const count = await this.app.redis.incr(key);
    if (count === 1 && ttl) {
      await this.app.redis.expire(key, ttl);
    }
    return count;
  }

  /**
   * 缓存用户权限
   */
  async cacheUserPermissions(
    userId: number,
    permissions: string[]
  ): Promise<void> {
    const key = `user:permissions:${userId}`;
    await this.app.redis.setex(key, 3600, JSON.stringify(permissions));
  }

  /**
   * 获取用户权限缓存
   */
  async getUserPermissions(userId: number): Promise<string[] | null> {
    const key = `user:permissions:${userId}`;
    const cached = await this.app.redis.get(key);
    return cached ? JSON.parse(cached) : null;
  }

  /**
   * 发布消息
   */
  async publish(channel: string, message: any): Promise<void> {
    await this.app.redis.publish(channel, JSON.stringify(message));
  }
}
```

## 十、定时任务

```typescript
// app/schedule/cleanExpiredToken.ts
import { Subscription } from 'egg';

export default class CleanExpiredToken extends Subscription {
  static get schedule() {
    return {
      interval: '1h',        // 每小时执行
      type: 'worker',        // 只在一个worker执行
      immediate: false,
      disable: false,
    };
  }

  async subscribe() {
    this.logger.info('[Schedule] 开始清理过期Token黑名单...');

    try {
      const pattern = `${this.app.config.appConfig.tokenBlacklistPrefix}*`;
      const keys = await this.app.redis.keys(pattern);

      let cleanedCount = 0;
      for (const key of keys) {
        const ttl = await this.app.redis.ttl(key);
        if (ttl <= 0) {
          await this.app.redis.del(key);
          cleanedCount++;
        }
      }

      this.logger.info(`[Schedule] 清理完成，共清理 ${cleanedCount} 个过期Token`);
    } catch (err) {
      this.logger.error('[Schedule] 清理Token失败', err);
    }
  }
}
```

```typescript
// app/extend/helper.ts
import * as crypto from 'crypto';

export default {
  /**
   * MD5加密
   */
  md5(str: string): string {
    return crypto.createHash('md5').update(str).digest('hex');
  },

  /**
   * 生成随机字符串
   */
  randomString(length: number = 16): string {
    return crypto.randomBytes(length).toString('hex').slice(0, length);
  },

  /**
   * 脱敏手机号
   */
  maskPhone(phone: string): string {
    return phone.replace(/(\d{3})\d{4}(\d{4})/, '$1****$2');
  },

  /**
   * 脱敏邮箱
   */
  maskEmail(email: string): string {
    const [name, domain] = email.split('@');
    const maskedName = name.slice(0, 2) + '***';
    return `${maskedName}@${domain}`;
  },

  /**
   * 格式化日期
   */
  formatDate(date: Date, format: string = 'YYYY-MM-DD HH:mm:ss'): string {
    const d = new Date(date);
    const map: Record<string, string> = {
      YYYY: d.getFullYear().toString(),
      MM: String(d.getMonth() + 1).padStart(2, '0'),
      DD: String(d.getDate()).padStart(2, '0'),
      HH: String(d.getHours()).padStart(2, '0'),
      mm: String(d.getMinutes()).padStart(2, '0'),
      ss: String(d.getSeconds()).padStart(2, '0'),
    };
    return format.replace(/YYYY|MM|DD|HH|mm|ss/g, (key) => map[key]);
  },
};
```
